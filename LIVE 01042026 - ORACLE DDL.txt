-- BLOCO 6 - Integridade: PRIMARY KEY + ALTER TABLE
---
 
CREATE TABLE N1
(CODIGO NUMBER(4),
NOME   VARCHAR2(30),
CONSTRAINT n1_codigo_pk PRIMARY KEY (codigo));
 
-- válido
INSERT INTO N1 VALUES (10,'TESTE');
 
-- erro esperado: PK duplicada
INSERT INTO N1 VALUES (10,'TESTE');
 
ALTER TABLE N1 ADD HOJE DATE;
ALTER TABLE N1 MODIFY HOJE VARCHAR2(30);
ALTER TABLE N1 MODIFY HOJE NUMBER;
 
INSERT INTO N1 VALUES (11,'TESTE', NULL);
 
ALTER TABLE N1 MODIFY HOJE DATE;
 
INSERT INTO N1 VALUES (12,'TESTE', SYSDATE);
 
ALTER TABLE N1 MODIFY HOJE NUMBER;
 
---
-- BLOCO 7 - Integridade referencial (FOREIGN KEY)
---
 
CREATE TABLE N2
(ID   NUMBER CONSTRAINT N2_ID_PK PRIMARY KEY,
NOME VARCHAR2(20) NOT NULL,
CODIGO NUMBER(4),
CONSTRAINT N1_CODIGO_FK FOREIGN KEY (CODIGO) REFERENCES N1(CODIGO));
 
-- válido
INSERT INTO N2 VALUES (1024, 'TESTE', 10);
 
-- erro esperado: chave estrangeira inexistente
INSERT INTO N2 VALUES (1025, 'TESTE', 20);
 
drop table n2 purge;
drop table n1 purge;

-- BLOCO 8 - CHECK CONSTRAINT
---
 
CREATE TABLE FORNECEDORES
(FORNECEDOR_ID NUMBER CONSTRAINT FORNECEDOR_ID_PK PRIMARY KEY,
FORNECEDOR_NOME VARCHAR2(20) NOT NULL,
STATUS NUMBER(1) CONSTRAINT STATUS_NN NOT NULL,
CATEGORIA VARCHAR2(20),
CONSTRAINT STATUS_CK CHECK (STATUS IN (4, 5)),
CONSTRAINT CATEGORIA_CK CHECK
(CATEGORIA IN ('Ativo','Inativo','Suspenso')));
